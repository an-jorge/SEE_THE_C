# Cálculo de Média Aritmética

```c
#include <stdio.h>

int main()
{
    int first_value = 10;
    int second_value = 8;
    int third_value = 5;
    int sum_of_values = first_value + second_value + third_value / 3;

    printf("Primeira nota: %d\n", first_value);
    printf("Segunda nota: %d\n", second_value);
    printf("Primeira nota: %d\n", third_value);

    printf("A MEDIA CALCULADA E %dv \n", sum_of_values);

    return 0;
}
```
